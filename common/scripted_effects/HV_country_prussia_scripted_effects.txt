PLS_holy_meeting_gui_update_effect = {
	if = {
		limit = { check_variable = { holy_meeting_gui_update > 10 } }
		subtract_from_variable = { holy_meeting_gui_update = 1 }
	}
	else = { add_to_variable = { holy_meeting_gui_update = 1 } }
}

PLS_holy_meeting_init_select_country = {
	clear_array = PLS_holy_meeting_select_country_array
	for_loop_effect = {
		end = 16
		
		add_to_array = { PLS_holy_meeting_select_country_array = 1 }
	}
	PLS_holy_meeting_gui_update_effect = yes
}

PLS_holy_meeting_country_select_AUS_effect = {
	for_each_loop = {
		array = PLS_holy_meeting_select_country_array
		index = i
		value = v
		
		if = {
			limit = { check_variable = { i = 0 } }
			set_variable = { PLS_holy_meeting_select_country_array^i = 2 }
			add_political_power = 22
		}
		else = { set_variable = { PLS_holy_meeting_select_country_array^i = 1 } }
	}
	PLS_holy_meeting_gui_update_effect = yes
}
