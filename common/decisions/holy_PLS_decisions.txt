	holy_debug_PLS_form_holy_roman_empire = {
		available = {
			is_debug = yes
			any_state = {
				has_state_flag = HV_is_heiliges_romisches_reich_state_flag
				NOT = { is_owned_by = PLS }
			}
		}
		fire_only_once = no
		complete_effect = {
			every_state = {
				limit = { 
					has_state_flag = HV_is_heiliges_romisches_reich_state_flag
					NOT = { is_owned_by = PLS }
				}
				if = {
					limit = { NOT = { is_core_of = PLS } }
					add_core_of = PLS
				}
				transfer_state_to = PLS
			}
			PLS = {
				set_cosmetic_tag = PLS_reich
			}
		}
	}
}
