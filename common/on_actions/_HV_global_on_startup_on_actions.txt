on_actions = {
	on_startup = {
		effect = {
			### initiate ruling coalition ###
			every_possible_country = {
				HV_init_party_entry = yes
				HV_lighten_ruling_party = yes
			}
			every_possible_country = {
				limit = {
					NOT = {
						tag = PLS
						tag = FRA
						tag = UKM
						tag = AUS
						has_country_flag = ROOT_has_country_leader_role_flag
					}
				}
				HV_init_party_leader = yes
			}
			UKM = {
				HV_global_init_countries_ruling_coalition = yes
			} # This is global setting, only need a country scope to trigger its effect, which tag is not important.
			every_possible_country = {
				HV_calc_ruling_coalition_popularity = yes
			}
			### initiate faction icon ###
			every_possible_country = {
				if = {
					limit = {
						OR = {
							tag = UKM
							is_in_faction_with = UKM
						}
					}
					set_variable = { faction_icon_var = 1 }
				}
				else_if = {
					limit = {
						OR = {
							tag = PLS
							is_in_faction_with = PLS
						}
					}
					set_variable = { faction_icon_var = 2 }
				}
			}
			### initiate welcome gui ###
			PLS = { set_variable = { HV_welcome_page = 3 } }
			UKM = { set_variable = { HV_welcome_page = 1 } }
			FRA = { set_variable = { HV_welcome_page = 1 } }
			every_possible_country = {
				add_to_array = { HV_welcome_container_tab_array = 1 }
				add_to_array = { HV_welcome_container_tab_array = 2 }
				add_to_array = { HV_welcome_container_tab_array = 1 }
				add_to_array = { HV_welcome_container_tab_array = 1 }
				if = {
					limit = { check_variable = { HV_welcome_page > 0 } }
					for_loop_effect = {
						end = HV_welcome_page
						value = v

						if = {
							limit = {
								check_variable = {
									v = 0
								}
							}
							add_to_array = { HV_welcome_tab_array = 2 }
						}
						else = {
							add_to_array = { HV_welcome_tab_array = 1 }
						}
					}
				}
				if = {
					limit = { is_ai = no }
					set_country_flag = ROOT_HV_welcome_gui_show_flag
				}
			}
		}
	}
}
